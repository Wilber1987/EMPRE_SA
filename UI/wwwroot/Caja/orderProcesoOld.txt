 * @property {Function} [action]
 * @property {Function} [SaveFunction]
 * @property {Orden_Movimiento} [entity]
 * @property {Denominacion_Catalogo[]} denominaciones
 * @property {{ Moneda: String, Moneda2: String }} [cambio_moneda]

export class WOrderProcessorComponent extends HTMLElement {
    /**
     * @param {WOrderProcessorComponentConfig} Config
     */
    constructor(Config) {
        super();

        this.style.backgroundColor = "#f8fafc";
        this.style.display = "flex";
        this.style.flexDirection = "column";
        this.style.height = "100%";
        this.style.overflow = "hidden";

        this.Config = Config;
        this.Entity = Config?.entity ?? new Orden_Movimiento();

        /** @type {Orden_Movimiento} */
        this.selectedOrder = this.Entity;
        this.liquidationMode = Config.liquidationMode ?? null;
        this.isFlexibleLiquidationModeActive = Config.isFlexibleLiquidationModeActive ?? (this.liquidationMode == null)
        this.targetUsd = new Money(0);
        this.targetNio = new Money(0);
        this.countedUsd = 0;
        this.countedNio = 0;

        /** @type {string|null} "USD" | "NIO" | null - Modo de vuelto cuando hay exceso de pago */
        //this.changeMode = null;
        this.changeMode = "USD";

        /**@type {Array<Denominacion_Catalogo>} */
        this.USD_DENOMS = Config?.denominaciones.filter(denom => denom.moneda == MonedaEnum.DOLARES) ?? [];
        /**@type {Array<Denominacion_Catalogo>} */
        this.NIO_DENOMS = Config?.denominaciones.filter(denom => denom.moneda == MonedaEnum.CORDOBAS) ?? [];

        /**@type {Array<Inventario_Denominacion_Caja>} */
        this.inventarioCajaOrigen = []
        /**@type {Array<Inventario_Denominacion_Caja>} */
        this.inventarioCajaDestino = []

        /** @type {Object.<string, Object.<number, number>>} */
        this.denomCounts = {
            USD: {},
            NIO: {}
        };
        this.InitDenomsCount();

        //this.InitAllDenomsCount();

        /** 
     * Contadores por bloque (contexto + moneda)
     * @type {Object.<string, Object.<number, number>>}
     */
        this.denomCountsByBlock = {
            payment_usd: {},
            payment_nio: {},
            exchange_usd: {},
            exchange_nio: {},
            change_usd: {},
            change_nio: {}
        };
        this.InitAllDenomsCount();

        this.DrawComponent();
    }

    InitAllDenomsCount() {
        const initBlock = (/** @type {string} */ blockKey, /** @type {any[]} */ denoms) => {
            this.denomCountsByBlock[blockKey] = {};
            denoms.forEach(d => this.denomCountsByBlock[blockKey][d.id_denominacion] = 0);
        };
        initBlock("payment_usd", this.USD_DENOMS);
        initBlock("payment_nio", this.NIO_DENOMS);
        initBlock("exchange_usd", this.USD_DENOMS);
        initBlock("exchange_nio", this.NIO_DENOMS);
        initBlock("change_usd", this.USD_DENOMS);
        initBlock("change_nio", this.NIO_DENOMS);
    }

    InitDenomsCount() {
        this.USD_DENOMS.forEach(d => this.denomCounts.USD[d.id_denominacion] = 0);
        this.NIO_DENOMS.forEach(d => this.denomCounts.NIO[d.id_denominacion] = 0);
    }

    connectedCallback() {
        this.addEventListener("keydown", this.handleTabTrap);
    }

    disconnectedCallback() {
        this.removeEventListener("keydown", this.handleTabTrap);
    }

    renderAllDenominationGrids = () => {
        const blocks = ["payment_usd", "payment_nio", "exchange_usd", "exchange_nio", "change_usd", "change_nio"];

        blocks.forEach(blockKey => {
            const container = this.querySelector(`#${blockKey}Container`);
            if (!container) return;

            const currency = blockKey.split("_")[1].toUpperCase();
            const denoms = currency === "USD" ? this.USD_DENOMS : this.NIO_DENOMS;
            const targetAmount = this.getTargetForBlock(blockKey);

            const targetDisplay = this.querySelector(`#target_${blockKey}_Display`);
            const countedDisplay = this.querySelector(`#counted_${blockKey}_Display`);
            const symbol = currency === "USD" ? "$" : "C$";
            const counted = this.calculateCountedForBlock(blockKey);

            if (this.liquidationMode != "MIXED" && blockKey.indexOf("change") !== 0) {
                if (targetDisplay) targetDisplay.textContent = `${symbol}${targetAmount.toFixed(2)}`;
            } else if (blockKey.indexOf("change") === 0) {
                if (targetDisplay) targetDisplay.textContent = `${symbol}${targetAmount.toFixed(2)}`;
            } else {
                if (targetDisplay) targetDisplay.textContent = `-`;
            }
            if (countedDisplay) countedDisplay.textContent = `${symbol}${counted.toFixed(2)}`;

            const grid = this.querySelector(`#${blockKey}Grid`);
            if (grid) {
                if (grid.children.length !== denoms.length) {
                    // Primera construcción
                    grid.innerHTML = "";
                    denoms.forEach(denom => {
                        const count = this.denomCountsByBlock[blockKey][denom.id_denominacion];
                        const subtotal = count * (denom.valor ?? 0);
                        grid.append(this.createDenomCard(blockKey, currency, denom, count, subtotal));
                    });
                } else {
                    // Actualización en sitio: no se destruye ningún botón ni input
                    const isTargetZero = targetAmount.toNumber() === 0;
                    denoms.forEach((denom, i) => {
                        const card = /** @type {HTMLElement} */ (grid.children[i]);
                        const count = this.denomCountsByBlock[blockKey][denom.id_denominacion];
                        const subtotal = count * (denom.valor ?? 0);

                        const input = /** @type {HTMLInputElement|null} */ (card.querySelector(".tp-denom-count"));
                        const subtotalEl = card.querySelector(".tp-denom-subtotal");

                        if (input && Number(input.value) !== count) input.value = String(count);
                        if (subtotalEl) subtotalEl.textContent = `${symbol}${subtotal.toFixed(2)}`;
                        card.classList.toggle("tp-denom-disabled", count === 0 && isTargetZero);
                    });
                }
            }
        });
    }

    GetTasaAplicada = () => {
        if (this.selectedOrder.direccion == DirectionEnum.MIXED) {
            return this.selectedOrder.moneda == MonedaEnum.CORDOBAS ? this.TasaActual?.Valor_de_venta : this.TasaActual?.Valor_de_compra
        }
        return this.selectedOrder.direccion == DirectionEnum.INGRESO
            ? this.TasaActual?.Valor_de_venta : this.TasaActual?.Valor_de_compra
    }

    /**
     * Obtiene el monto requerido para un bloque específico
     */
    getTargetForBlock = (/** @type {string} */ blockKey) => {
        // Manejar bloques de vuelto (change)
        if (blockKey === "change_usd" || blockKey === "change_nio") {
            return this.getTargetForChangeBlock(blockKey);
        }

        const isOrderUsd = this.selectedOrder.moneda === MonedaEnum.DOLARES;
        const orderAmount = new Money(this.selectedOrder?.monto) ?? new Money(0);
        const exchangeRateCompra = this.TasaActual?.Valor_de_compra ?? 0;
        const exchangeRateVenta = this.TasaActual?.Valor_de_venta ?? 0;

        const tasa = (this.GetTasaAplicada()) ?? 0

        if (this.liquidationMode === "USD") {
            if (blockKey === "payment_usd") return isOrderUsd ? orderAmount : orderAmount.divide(tasa);
            if (blockKey === "exchange_nio") return isOrderUsd ? new Money(0) : orderAmount;
            return new Money(0);
        }
        else if (this.liquidationMode === "NIO") {
            if (blockKey === "payment_nio") return isOrderUsd ? orderAmount.multiply(tasa) : orderAmount;
            if (blockKey === "exchange_usd") return isOrderUsd ? orderAmount : new Money(0);
            return new Money(0);
        }
        else if (this.liquidationMode === "MIXED_PAYMENT") {
            if (isOrderUsd) {
                if (blockKey === "payment_usd") return this.targetUsd;
                if (blockKey === "payment_nio") return this.targetNio;
                if (blockKey === "exchange_usd") return this.targetNio.divide(tasa);
                return new Money(0);
            } else {
                if (blockKey === "payment_nio") return this.targetNio;
                if (blockKey === "payment_usd") return this.targetUsd;
                if (blockKey === "exchange_nio") return this.targetUsd.multiply(tasa);
                return new Money(0);
            }
        }
        else if (this.liquidationMode === "MIXED") {
            if (isOrderUsd) {
                if (blockKey === "exchange_usd") return this.targetNio.divide(tasa);
                return new Money(0);
            } else {
                if (blockKey === "exchange_nio") return this.targetUsd.multiply(tasa);
                return new Money(0);
            }
        }
        return new Money(0);
    }

    /**
     * Calcula el target para bloques de vuelto (change)
     * @param {string} blockKey
     */
    getTargetForChangeBlock = (blockKey) => {
        if (this.changeMode == null || !this.hasPaymentExcess()) {
            return new Money(0);
        }

        const isOrderUsd = this.selectedOrder.moneda === MonedaEnum.DOLARES;
        const tasaCompra = new Money(this.TasaActual?.Valor_de_compra ?? 0);
        const tasaVenta = new Money(this.TasaActual?.Valor_de_venta ?? 0);

        // Exceso expresado en la moneda de la orden
        const excess = this.calculatePaymentExcess();

        if (blockKey === "change_usd") {
            if (this.changeMode !== "USD") return new Money(0);
            // Orden USD + vuelto USD: exceso directo
            if (isOrderUsd) return excess;
            // Orden NIO + vuelto USD: exceso / valor de venta
            return excess.divide(tasaVenta);
        }

        if (blockKey === "change_nio") {
            if (this.changeMode !== "NIO") return new Money(0);
            // Orden USD + vuelto NIO: exceso * valor de compra
            if (isOrderUsd) return excess.multiply(tasaCompra);
            // Orden NIO + vuelto NIO: exceso directo
            return excess;
        }

        return new Money(0);
    }

    /**
     * Calcula el exceso de pago (cuando las denominaciones superan el monto de la orden)
     */
    calculatePaymentExcess = () => {
        if (this.selectedOrder?.direccion !== DirectionEnum.INGRESO) {
            return new Money(0);
        }

        const isOrderUsd = this.selectedOrder.moneda === MonedaEnum.DOLARES;
        const orderAmount = new Money(this.selectedOrder?.monto ?? 0);
        const tasaCompra = new Money(this.TasaActual?.Valor_de_compra ?? 0);
        const tasaVenta = new Money(this.TasaActual?.Valor_de_venta ?? 0);

        const paymentUsd = this.calculateCountedForBlock("payment_usd");
        const paymentNio = this.calculateCountedForBlock("payment_nio");

        let totalPaidInOrderCurrency;
        if (isOrderUsd) {
            // NIO entrando a caja sobre orden USD => tasa de venta
            totalPaidInOrderCurrency = paymentUsd.add(paymentNio.divide(tasaVenta));
        } else {
            // USD entrando a caja sobre orden NIO => tasa de compra
            totalPaidInOrderCurrency = paymentNio.add(paymentUsd.multiply(tasaCompra));
        }

        return totalPaidInOrderCurrency.subtract(orderAmount);
    }

    /**
     * Verifica si hay exceso de pago que requiera vuelto
     */
    hasPaymentExcess = () => {
        return this.calculatePaymentExcess().toNumber() > 0.01;
    }

    /**
     * Calcula el total contado en un bloque específico
     */
    calculateCountedForBlock = (/** @type {string} */ blockKey) => {
        const currency = blockKey.split("_")[1].toUpperCase();
        const denoms = currency === "USD" ? this.USD_DENOMS : this.NIO_DENOMS;
        return denoms.reduce((sum, d) => {
            return new Money(sum).add(new Money(d.valor ?? 0).multiply(this.denomCountsByBlock[blockKey][d.id_denominacion]))
        }, new Money(0));
    }

        }
    }

    /**

   */

        const colorClass = currency === "USD" ? "tp-color-green" : "tp-color-blue";
        const contextClass = context === "exchange" ? "tp-currency-exchange" : context === "change" ? "tp-currency-change" : "tp-currency-payment";

                            children: [
                                WRender.Create({ className: `tp-color-dot ${colorClass}` }),
                                WRender.Create({ tagName: "h4", innerText: label })

    /**
     * @param {string} mode - "USD" | "NIO" | "MIXED_PAYMENT"
     */
    setLiquidationMode = (mode) => {
        this.liquidationMode = mode;
        this.changeMode = "USD"; // Resetear modo de vuelto al cambiar método de liquidación
        // 🔥 NUEVO: Limpiar los bloques que cambian de propósito
        this.resetBlockCounts("payment_usd");
        this.resetBlockCounts("payment_nio");
        this.resetBlockCounts("exchange_usd");
        this.resetBlockCounts("exchange_nio");
        this.resetBlockCounts("change_usd");
        this.resetBlockCounts("change_nio");


        this.querySelectorAll(".tp-liquidation-btn").forEach(btn => {
            btn.classList.remove("tp-liquidation-btn-active");
        });
        const activeBtn = this.querySelector(`#btnMode${mode}`);
        if (activeBtn) activeBtn.classList.add("tp-liquidation-btn-active");

        const isOrderUsd = this.selectedOrder.moneda === MonedaEnum.DOLARES;
        const orderAmount = new Money(this.selectedOrder?.monto ?? 0);
        const tasa = this.GetTasaAplicada();

        const exchangeRate = new Money(tasa ?? 0);

        if (mode === "USD") {
            if (isOrderUsd) {
                this.targetUsd = orderAmount;
                this.targetNio = new Money(0);
                this.selectedOrder.tipo_operacion_cambio = "NONE";
            } else {
                this.targetUsd = orderAmount.divide(exchangeRate);
                this.targetNio = new Money(0);
                this.selectedOrder.tipo_operacion_cambio = "COMPRA_MONEDA";
                this.selectedOrder.monto_cambio_usd = this.targetUsd.toNumber();
                this.selectedOrder.monto_cambio_nio = orderAmount.toNumber();
                this.selectedOrder.tasa_cambio_aplicada = exchangeRate.toNumber();
            }
        }
        else if (mode === "NIO") {
            if (isOrderUsd) {
                this.targetUsd = new Money(0);
                this.targetNio = orderAmount.multiply(exchangeRate);
                this.selectedOrder.tipo_operacion_cambio = "VENTA_MONEDA";
                this.selectedOrder.monto_cambio_usd = orderAmount.toNumber();
                this.selectedOrder.monto_cambio_nio = this.targetNio.toNumber();
                this.selectedOrder.tasa_cambio_aplicada = exchangeRate.toNumber();
            } else {
                this.targetUsd = new Money(0);
                this.targetNio = orderAmount;
                this.selectedOrder.tipo_operacion_cambio = "NONE";
            }
        }
        else if (mode === "MIXED_PAYMENT") {
            const mixedInputs = this.querySelector("#mixedInputs");
            if (mixedInputs) mixedInputs.classList.remove("hidden");

            const mixedUsdAmount = this.querySelector("#mixedUsdAmount");
            const mixedNioAmount = this.querySelector("#mixedNioAmount");
            if (mixedUsdAmount) /** @type {HTMLInputElement} */ (mixedUsdAmount).value = "";
            if (mixedNioAmount) /** @type {HTMLInputElement} */ (mixedNioAmount).value = "";

            this.targetUsd = new Money(0);
            this.targetNio = new Money(0);
            this.selectedOrder.tipo_operacion_cambio = "MIXED_PAYMENT";
        } else if (mode === "MIXED") {
            const isCompra = this.selectedOrder.direccion == DirectionEnum.MIXED
                && this.selectedOrder.moneda == MonedaEnum.DOLARES;

            this.selectedOrder.tipo_operacion_cambio = isCompra
                ? "COMPRA_MONEDA" : "VENTA_MONEDA";
        }

        const mixedInputs = this.querySelector("#mixedInputs");
        if (mode !== "MIXED_PAYMENT" && mixedInputs) mixedInputs.classList.add("hidden");

        const denominationsSection = this.querySelector("#denominationsSection");
        if (denominationsSection) denominationsSection.classList.remove("hidden");
        // Dentro de setLiquidationMode(), al final, antes de renderDenominationBlocks:
        if (this.selectedOrder?.direccion === DirectionEnum.EGRESO) {
            if (mode === "USD" && isOrderUsd) {
                this.applyOptimalCombinationToBlock("payment_usd", this.targetUsd);
            } else if (mode === "NIO" && !isOrderUsd) {
                this.applyOptimalCombinationToBlock("payment_nio", this.targetNio);
            } else if (mode === "USD" && !isOrderUsd) {
                // Caja entrega USD por una orden NIO (cambio)
                this.applyOptimalCombinationToBlock("exchange_usd", this.getTargetForBlock("exchange_usd"));
            } else if (mode === "NIO" && isOrderUsd) {
                this.applyOptimalCombinationToBlock("exchange_nio", this.getTargetForBlock("exchange_nio"));
            }
        }

        this.renderDenominationBlocks();
        this.updateValidation();
    }

    /**
    * @param {string} blockKey - "payment_usd", "exchange_nio", etc.
    * @param {string} currency
    * @param {Denominacion_Catalogo} denom
    * @param {number} count
    * @param {number} subtotal
    */
    createDenomCard = (blockKey, currency, denom, count, subtotal) => {
        const targetAmount = this.getTargetForBlock(blockKey);
        const isDisabled = count === 0 && targetAmount.toNumber() === 0;
        const opacityClass = isDisabled ? "tp-denom-disabled" : "";
        const symbol = currency === "USD" ? "$" : "C$";
        const direction = this.selectedOrder.direccion;
        console.log(direction);
        const validateIngreso = this.selectedOrder.direccion == DirectionEnum.INGRESO;
        const validateEgreso = this.selectedOrder.direccion == DirectionEnum.EGRESO;

        /**  payment_usd, exchange_nio, payment_nio, exchange_usd, change_usd, change_nio */
        /**@type {Inventario_Denominacion_Caja[]} */
        const inventory = [];
        let hiddenDenom = false;
        /**@type {Inventario_Denominacion_Caja | undefined} */
        let findInventory;
        const findInventoryFun = () => {
            findInventory = inventory.find(i => i.Denominacion_Catalogo?.valor == denom.valor
                && i.Denominacion_Catalogo?.moneda == denom.moneda
                && i.Denominacion_Catalogo?.tipo == denom.tipo);
        }
        if (["payment_usd", "payment_nio", "exchange_nio", "exchange_usd"].includes(blockKey) && validateEgreso) {
            inventory.length = 0;
            inventory.push(...this.inventarioCajaOrigen)
            findInventoryFun();
            hiddenDenom = (validateEgreso && (findInventory == null || findInventory.cantidad_actual == 0))
        } if (["change_usd", "change_nio"].includes(blockKey) && validateIngreso) {
            inventory.length = 0;
            inventory.push(...this.inventarioCajaDestino)
            findInventoryFun();
            hiddenDenom = (validateIngreso && (findInventory == null || findInventory.cantidad_actual == 0))
        }

        const denomWrapper = html`<div class="tp-denom-card ${opacityClass}">
            <span class="tp-denom-value">${symbol}${denom.valor}</span>
            <div class="tp-denom-controls">
                <button tabindex="-1" min="0" onclick="${() => this.updateDenom(blockKey, currency, denom, -1, true)}" 
                    class="tp-denom-btn tp-denom-btn-minus">-</button>
                <input min="0" max="${findInventory?.cantidad_actual ?? ""}" type="number" value="${count}" 
                    onchange="${(/** @type {{ target: { value: number; }; }} */ ev) => this.updateDenom(blockKey, currency, denom, ev.target.value)}" 
                    class="tp-denom-count" />
                <button tabindex="-1" min="0" onclick="${() => this.updateDenom(blockKey, currency, denom, 1, true)}" 
                    class="tp-denom-btn tp-denom-btn-plus">+</button>
            </div>
            <span class="tp-denom-subtotal">${symbol}${subtotal.toFixed(2)}</span>
        </div>`;
        denomWrapper.style.display = hiddenDenom ? "none" : "grid";
        return denomWrapper;
    }

    /**
     * @param {string} blockKey
     * @param {string} currency
     * @param {Denominacion_Catalogo} denom
     * @param {number} change
     */
    updateDenom = (blockKey, currency, denom, change, isButtom = false) => {
        if (change < 0 && !isButtom) {
            change = 0;
        }
        const newCount = !isButtom ? change
            : parseInt(this.denomCountsByBlock[blockKey][denom.id_denominacion].toString()) + parseInt(change.toString());
        if (newCount < 0) return;

        this.denomCountsByBlock[blockKey][denom.id_denominacion] = newCount;

        // >>> NUEVO: mostrar/ocultar sección de vuelto si cambió el exceso <<<
        this.syncChangeSection();

        this.renderAllDenominationGrids();
        this.updateValidation();
    }

    calculateTotals = () => {
        this.countedUsd = this.USD_DENOMS.reduce((sum, d) => sum + (this.denomCounts.USD[d.id_denominacion] * (d.valor ?? 0)), 0);
        this.countedNio = this.NIO_DENOMS.reduce((sum, d) => sum + (this.denomCounts.NIO[d.id_denominacion] * (d.valor ?? 0)), 0);

        const countedUsdDisplay = this.querySelector("#countedUSDDisplay");
        const countedNioDisplay = this.querySelector("#countedNIODisplay");

        if (countedUsdDisplay) countedUsdDisplay.textContent = `$${this.countedUsd.toFixed(2)}`;
        if (countedNioDisplay) countedNioDisplay.textContent = `C$${this.countedNio.toFixed(2)}`;
    }

    updateValidation = () => {
        const statusDiv = this.querySelector("#validationStatus");
        const statusIcon = this.querySelector("#statusIcon");
        const statusTitle = this.querySelector("#statusTitle");
        const statusDesc = this.querySelector("#statusDesc");
        const processBtn = this.querySelector("#processBtn");

        if (!statusDiv || !statusIcon || !statusTitle || !statusDesc || !processBtn) return;

        const activeBlocks = this.getActiveBlocks();
        let allValid = true;
        let messages = [];

        for (const blockKey of activeBlocks) {
            if (this.liquidationMode == "MIXED") {
                if (blockKey == "exchange_usd") {
                    this.targetUsd = this.calculateCountedForBlock(blockKey);
                } else if (blockKey == "exchange_nio") {
                    this.targetNio = this.calculateCountedForBlock(blockKey);
                }
                continue;
            }
            const target = this.getTargetForBlock(blockKey);
            const counted = this.calculateCountedForBlock(blockKey);
            const diff = counted.subtract(target);
            const currency = blockKey.split("_")[1].toUpperCase();
            const symbol = currency === "USD" ? "$" : "C$";

            let context = "";
            if (blockKey.startsWith("payment")) context = "Pago";
            else if (blockKey.startsWith("exchange")) context = "Cambio";
            else if (blockKey.startsWith("change")) context = "Vuelto";

            const isPaymentBlock = blockKey.startsWith("payment");
            const hasActiveChange = this.changeMode != null
                && this.selectedOrder?.direccion === DirectionEnum.INGRESO;

            if (isPaymentBlock && hasActiveChange) {
                // Cuando hay vuelto activo, el pago puede SUPERAR el monto.
                // Solo validar que cubra al menos el monto requerido.
                if (counted.toNumber() < target.toNumber() - 0.01) {
                    allValid = false;
                    messages.push(`${context} ${currency}: falta ${symbol}${target.subtract(counted).toFixed(2)}`);
                }
            } else {
                // Validación estricta: debe coincidir exactamente
                if (Math.abs(diff.toNumber()) > 0.01) {
                    allValid = false;
                    messages.push(`${context} ${currency}: diferencia ${diff.toNumber() > 0 ? "+" : ""}${symbol}${diff.toFixed(2)}`);
                }
            }
        }

        if (this.liquidationMode === "MIXED_PAYMENT") {
            const exchangeRate = this.GetTasaAplicada();
            const isOrderUsd = this.selectedOrder.moneda === MonedaEnum.DOLARES;
            const orderAmount = new Money(this.selectedOrder?.monto) ?? new Money(0);

            let totalInOrderCurrency;
            if (isOrderUsd) {
                totalInOrderCurrency = new Money(this.targetUsd)
                    .add(new Money(this.targetNio)
                        .divide(exchangeRate));
            } else {
                totalInOrderCurrency = new Money(this.targetNio)
                    .add(new Money(this.targetUsd)
                        .multiply(exchangeRate));
            }

            if (Math.abs(totalInOrderCurrency.toNumber() - orderAmount.toNumber()) > 0.01) {
                allValid = false;
                messages.push(`Distribución mixta incorrecta`);
            }
        }

        if (this.liquidationMode === "MIXED") {
            const exchangeRate = this.GetTasaAplicada();
            const isOrderUsd = this.selectedOrder.moneda === MonedaEnum.DOLARES;

            let totalUsdEsperadoConvertido = new Money(this.targetNio)
                .divide(exchangeRate);
            let totalNioEsperadoConvertido = new Money(this.targetUsd)
                .multiply(exchangeRate);

            if (!isOrderUsd && Math.abs(totalUsdEsperadoConvertido.toNumber() - this.targetUsd.toNumber()) > 0.01) {
                allValid = false;
                messages.push(`Distribución de cambio incorrecto, tasa: ${exchangeRate}, 
                    esperado ${totalUsdEsperadoConvertido}, faltante: ${totalUsdEsperadoConvertido.subtract(this.targetUsd)}`);
            }
            if (isOrderUsd && Math.abs(totalNioEsperadoConvertido.toNumber() - this.targetNio.toNumber()) > 0.01) {
                allValid = false;
                messages.push(`Distribución de cambio incorrecto, tasa: ${exchangeRate}, 
                    esperado ${totalNioEsperadoConvertido}, faltante: ${totalNioEsperadoConvertido.subtract(this.targetNio)}`);
            }
        }
        if (this.selectedOrder?.direccion === DirectionEnum.INGRESO
            && this.hasPaymentExcess()
            && this.changeMode == null) {
            allValid = false;
            messages.push("Seleccione el tipo de cambio para el excedente de pago");
        }

        if (allValid && this.liquidationMode !== null && activeBlocks.length > 0) {
            this.setValidationState(true, "Monto correcto", "Todas las denominaciones coinciden exactamente.", statusDiv, statusIcon, statusTitle, statusDesc, processBtn);
        } else {
            const msg = messages.length > 0 ? messages.join(". ") + "." : "Revise las denominaciones ingresadas.";
            this.setValidationState(false, "Monto no coincide", msg, statusDiv, statusIcon, statusTitle, statusDesc, processBtn);
        }
    }

    /**
     * Retorna los bloques que están activos según el modo de liquidación
     */
    getActiveBlocks = () => {
        const isOrderUsd = this.selectedOrder.moneda === MonedaEnum.DOLARES;

        /**
         * @type {string[]}
         */
        let blocks = [];

        if (this.liquidationMode === "USD") {
            blocks = isOrderUsd ? ["payment_usd"] : ["payment_usd", "exchange_nio"];
        }
        else if (this.liquidationMode === "NIO") {
            blocks = isOrderUsd ? ["payment_nio", "exchange_usd"] : ["payment_nio"];
        }
        else if (this.liquidationMode === "MIXED_PAYMENT") {
            blocks = isOrderUsd
                ? ["payment_usd", "payment_nio"]
                : ["payment_nio", "payment_usd"];
        }
        else if (this.liquidationMode === "MIXED") {
            blocks = ["exchange_usd", "exchange_nio"];
        }

        // Agregar bloques de vuelto si hay exceso de pago y se ha seleccionado un modo
        if (this.selectedOrder?.direccion === DirectionEnum.INGRESO && this.hasPaymentExcess()) {
            if (this.changeMode === "USD") {
                blocks.push("change_usd");
            } else if (this.changeMode === "NIO") {
                blocks.push("change_nio");
            }
        }

        return blocks;
    }

    processTransaction = async () => {
        if (!this.selectedOrder) return;

        this.append(ModalVericateAction(async () => {
            const isOrderUsd = this.selectedOrder.moneda === MonedaEnum.DOLARES;

            this.selectedOrder.tipo_vuelto = this.changeMode;
            this.selectedOrder.DenominationsUSD = this.buildDenominationsFromBlock("payment_usd", this.selectedOrder.direccion);
            this.selectedOrder.DenominationsNIO = this.buildDenominationsFromBlock("payment_nio", this.selectedOrder.direccion);

            const exchangeDirection = this.selectedOrder.direccion === DirectionEnum.INGRESO
                ? DirectionEnum.EGRESO
                : DirectionEnum.INGRESO;

            this.selectedOrder.DenominationsCambioUSD = this.buildDenominationsFromBlock("exchange_usd", exchangeDirection);
            this.selectedOrder.DenominationsCambioNIO = this.buildDenominationsFromBlock("exchange_nio", exchangeDirection);

            // Agregar denominaciones de vuelto (vuelto al cliente, dirección inversa al pago)
            const changeDirection = this.selectedOrder.direccion === DirectionEnum.INGRESO
                ? DirectionEnum.EGRESO
                : DirectionEnum.INGRESO;

            this.selectedOrder.DenominationsVueltoUSD = this.buildDenominationsFromBlock("change_usd", changeDirection);
            this.selectedOrder.DenominationsVueltoNIO = this.buildDenominationsFromBlock("change_nio", changeDirection);

            if (this.Config.SaveFunction) {
                this.Config.SaveFunction(this.selectedOrder);
                return;
            }
            const response = await this.selectedOrder.AprobarOrden();
            if (response.status == 200) {
                if (this.Config?.action) {
                    this.Config.action(this.selectedOrder);
                    return;
                }
                this.showSuccessModal();
            } else {
                WAlertMessage.Danger(response.message, true);
            }
        }, "¿Está seguro de ejecutar este proceso?"))
    }

    /**
     * Construye el array de DTOs desde un bloque específico
     * @param {string} blockKey
     * @param {string?} direction
     * @returns {DenominacionMovimientoDTO[]}
     */
    buildDenominationsFromBlock = (blockKey, direction) => {
        const currency = blockKey.split("_")[1].toUpperCase();
        const counts = this.denomCountsByBlock[blockKey];

        return Object.keys(counts).map(key => {
            const cantidad = counts[parseInt(key)];
            if (cantidad === 0) return null;

            return new DenominacionMovimientoDTO({
                id_denominacion: parseInt(key),
                tipo_operacion: direction,
                cantidad: cantidad
            });
        }).filter(d => d !== null);
    }

    /**

    syncChangeSection = () => {
        const shouldShow = this.selectedOrder?.direccion === DirectionEnum.INGRESO && this.hasPaymentExcess();
        const isShown = !!this.querySelector("#changeModeSection");

        if (shouldShow === isShown) return;

        if (!shouldShow) {

            this.changeMode = null;
            this.resetBlockCounts("change_usd");
            this.resetBlockCounts("change_nio");

        // 🔥 NUEVO: Si acaba de aparecer la sección de vuelto, calcular automáticamente
        if (shouldShow && !isShown && this.changeMode) {
            const targetBlock = this.changeMode === "USD" ? "change_usd" : "change_nio";

    captureFocus = () => {
        const active = /** @type {HTMLElement|null} */ (document.activeElement);
        if (!active || !this.contains(active)) return null;
        const card = active.closest(".tp-denom-card");
        const grid = card?.closest(".tp-denom-grid");
        if (!card || !grid) return null;

        const selector = active.classList.contains("tp-denom-btn-plus") ? ".tp-denom-btn-plus"
            : active.classList.contains("tp-denom-btn-minus") ? ".tp-denom-btn-minus"
                : ".tp-denom-count";

        return { gridId: grid.id, index: Array.from(grid.children).indexOf(card), selector };
    }

        if (!info) return;
        const el = /** @type {HTMLElement|null} */ (
            this.querySelector(`#${info.gridId}`)?.children[info.index]?.querySelector(info.selector)
        );
        el?.focus();
    }

    handleTabTrap = (/** @type {KeyboardEvent} */ ev) => {
        if (ev.key !== "Tab") return;

        const tabbables = this.getTabbables();
        if (tabbables.length === 0) {
            ev.preventDefault();
            return;
        }

        const first = tabbables[0];
        const last = tabbables[tabbables.length - 1];
        const active = /** @type {HTMLElement} */ (document.activeElement);

        // Si el foco se perdió (body), dejamos que el navegador decida
        if (!active || !this.contains(active)) return;

        const FOLLOWING = Node.DOCUMENT_POSITION_FOLLOWING;
        const PRECEDING = Node.DOCUMENT_POSITION_PRECEDING;

        if (!ev.shiftKey) {
            // Tab: ¿estoy en el último, o en un elemento (ej. botón +) situado después del último?
            const isAtEnd = active === last || !!(last.compareDocumentPosition(active) & FOLLOWING);
            if (isAtEnd) {
                ev.preventDefault();
                first.focus();
                if (first instanceof HTMLInputElement) first.select();
            }
        } else {
            // Shift+Tab: ¿estoy en el primero, o antes del primero?
            const isAtStart = active === first || !!(first.compareDocumentPosition(active) & PRECEDING);
            if (isAtStart) {
                ev.preventDefault();
                last.focus();
                if (last instanceof HTMLInputElement) last.select();
            }
        }
    }
    /**
     * Elementos que realmente reciben foco con Tab dentro del componente
     * (excluye tabindex="-1", deshabilitados y ocultos)
     * @returns {HTMLElement[]}
     */
    getTabbables = () => {
        const selector = 'input, select, textarea, button, a[href], [tabindex]:not([tabindex="-1"])';
        return /** @type {HTMLElement[]} */ (Array.from(this.querySelectorAll(selector))).filter(el => {
            if (el.getAttribute("tabindex") === "-1") return false;
            if (/** @type {any} */ (el).disabled) return false;
            // oculto (display:none o dentro de un contenedor .hidden)
            return el.getClientRects().length > 0;
        });
    }

    //bundary
    /**
 * Calcula la combinación óptima de denominaciones para alcanzar un monto exacto,
 * minimizando el número total de billetes/monedas, respetando el inventario.
 * Usa Programación Dinámica (bounded knapsack).
 *
 * @param {Money} targetAmount - Monto objetivo
 * @param {Inventario_Denominacion_Caja[]} inventory - Inventario disponible
 * @param {string} currency - "USD" | "NIO"
 * @returns {Object.<number, number>|null} - Map { id_denominacion: cantidad } o null si no es posible
 */
    calculateOptimalCombination = (targetAmount, inventory, currency) => {
        const targetCents = Math.round(targetAmount.toNumber() * 100);
        if (targetCents === 0) return {};

        // 1. Preparar denominaciones disponibles
        const monedaEnum = currency === "USD" ? MonedaEnum.DOLARES : MonedaEnum.CORDOBAS;
        const denoms = inventory
            .filter(i => i.Denominacion_Catalogo?.moneda === monedaEnum)
            .map(i => ({
                id: i.Denominacion_Catalogo?.id_denominacion,
                valueCents: Math.round((i.Denominacion_Catalogo?.valor ?? 0) * 100),
                maxCount: i.cantidad_actual ?? 0
            }))
            .filter(d => d.valueCents > 0 && d.maxCount > 0)
            .sort((a, b) => b.valueCents - a.valueCents); // Mayor a menor

        if (denoms.length === 0) return null;

        // 2. DP: dp[amount] = { minCoins, combination }
        // Usamos un Map para evitar array gigante si targetCents es grande
        const dp = new Map();
        dp.set(0, { minCoins: 0, combo: new Map() });

        for (const denom of denoms) {
            const { id, valueCents, maxCount } = denom;
            // Clonamos los estados actuales para no reutilizar la misma denominación en la misma iteración
            const states = Array.from(dp.entries());
            for (const [currentAmount, state] of states) {
                for (let k = 1; k <= maxCount; k++) {
                    const newAmount = currentAmount + k * valueCents;
                    if (newAmount > targetCents) break;
                    const newCoins = state.minCoins + k;
                    const existing = dp.get(newAmount);
                    if (!existing || newCoins < existing.minCoins) {
                        const newCombo = new Map(state.combo);
                        newCombo.set(id, (newCombo.get(id) ?? 0) + k);
                        dp.set(newAmount, { minCoins: newCoins, combo: newCombo });
                    }
                }
            }
        }

        const result = dp.get(targetCents);
        if (!result) return null; // No hay combinación exacta posible

        /**@type {Object.<string, any>} */
        const output = {};
        result.combo.forEach((/** @type {any} */ count, /** @type {string | number} */ id) => { output[id] = count; });
        return output;
    };

    /**
     * Aplica la combinación óptima a un bloque específico.
     * @param {string} blockKey
     * @param {Money} target
     */
    applyOptimalCombinationToBlock = (blockKey, target) => {
        const currency = blockKey.split("_")[1].toUpperCase();
        // En egresos y vueltos se usa inventario; aquí elegimos según contexto
        const inventory = this.selectedOrder?.direccion === DirectionEnum.EGRESO
            ? this.inventarioCajaOrigen
            : this.inventarioCajaDestino;

        const combination = this.calculateOptimalCombination(target, inventory, currency);
        if (!combination) {
            WAlertMessage.Warning(
                `No es posible cubrir exactamente ${currency} $${target.toFixed(2)} con el inventario actual.`,
                true
            );
            return;
        }

        // Resetear el bloque y aplicar la combinación
        this.resetBlockCounts(blockKey);
        for (const [idStr, count] of Object.entries(combination)) {
            this.denomCountsByBlock[blockKey][parseInt(idStr)] = count;
        }

        this.renderAllDenominationGrids();
        this.updateValidation();
    };
}
